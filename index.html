スプレッドシート設定 & GASコード

※下記GASスクリプトをデプロイして発行されたURLを貼り付けてください。

設定手順ガイド

  1. Google スプレッドシートを新規作成。シート名を「注文一覧」に変更。
  2. メニューの **「拡張機能」＞「Apps Script」** を開く。
  3. 既存のコードを消去し、下のスクリプトをそのまま貼り付け。
  4. 右上の **「デプロイ」＞「新しいデプロイ」** を選択。
  5. 種類で **「Web アプリ」** を選択し、アクセスできるユーザーを **「全員（Anyone）」** に設定してデプロイ。
  6. 発行された **Web アプリ URL** を上のフォームに貼り付けて保存。
GASコード (コピーしてApps Scriptに貼り付け) 
function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 「注文一覧」シートを取得（無ければ現在のシート）
    let sheet = ss.getSheetByName("注文一覧") || ss.getActiveSheet();
    
    // 「集計」シートの準備（自動生成）
    let summarySheet = ss.getSheetByName("集計");
    if (!summarySheet) {
      summarySheet = ss.insertSheet("集計");
      summarySheet.getRange("A1").setValue("累計販売食数");
      summarySheet.getRange("B1").setValue(0);
    }

    const data = JSON.parse(e.postData.contents);
    const lastRow = sheet.getLastRow();
    
    let deletedCount = 0; // 今回削除する完了済みデータの件数

    // 1. チェックボックス（E列 = 5列目）がTRUEの行を下から逆順に一括削除
    for (let r = lastRow; r >= 2; r--) {
      const isChecked = sheet.getRange(r, 5).getValue();
      if (isChecked === true || isChecked === "TRUE") {
        deletedCount++;
        sheet.deleteRow(r);
      }
    }

    // 2. 「集計」シートの累計販売食数（B1）に加算
    if (deletedCount > 0) {
      const currentTotal = Number(summarySheet.getRange("B1").getValue()) || 0;
      summarySheet.getRange("B1").setValue(currentTotal + deletedCount);
    }

    // 3. 今回届いた新しい注文データを書き込み
    const now = new Date();
    data.items.forEach(item => {
      sheet.appendRow([
        now,          // A列: タイムスタンプ
        item.name,    // B列: 種類 (もも, ねぎま, つくね)
        item.flavor,  // C列: 味 (たれ, 塩)
        item.topping, // D列: トッピング
        false         // E列: チェックボックス (デフォルト未チェック)
      ]);
    });

    // 4. 1セット（1客分）終わったら1行空ける
    sheet.appendRow([]);

    return ContentService.createTextOutput(JSON.stringify({ status: "success", deletedCount: deletedCount }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

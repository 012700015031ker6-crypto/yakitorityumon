function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("注文一覧") || ss.getActiveSheet();
    let summarySheet = ss.getSheetByName("集計");
    if (!summarySheet) {
      summarySheet = ss.insertSheet("集計");
      summarySheet.getRange("A1").setValue("累計販売食数");
      summarySheet.getRange("B1").setValue(0);
    }
    const data = JSON.parse(e.postData.contents);
    const lastRow = sheet.getLastRow();
    let deletedCount = 0;
    for (let r = lastRow; r >= 2; r--) {
      const isChecked = sheet.getRange(r, 5).getValue();
      if (isChecked === true || isChecked === "TRUE") {
        deletedCount++;
        sheet.deleteRow(r);
      }
    }
    if (deletedCount > 0) {
      const currentTotal = Number(summarySheet.getRange("B1").getValue()) || 0;
      summarySheet.getRange("B1").setValue(currentTotal + deletedCount);
    }
    const now = new Date();
    data.items.forEach(item => {
      sheet.appendRow([now, item.name, item.flavor, item.topping, false]);
    });
    sheet.appendRow([]);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", deletedCount: deletedCount }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
